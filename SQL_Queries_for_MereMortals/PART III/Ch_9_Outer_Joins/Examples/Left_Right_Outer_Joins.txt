-- "Show me all the recipe types and any matching recipes in my database." 

SELECT Recipe_Classes.RecipeClassDescription,
	Recipes.RecipeTitle
FROM Recipe_Classes 
LEFT OUTER JOIN Recipes 
ON Recipe_Classes.RecipeClassID = 
	Recipes.RecipeClassID
;

-- "List all the recipe classes that do not yet have any recipes." 

SELECT Recipe_Classes.RecipeClassDescription
FROM Recipe_Classes
LEFT OUTER JOIN Recipes 
ON 
Recipe_Classes.RecipeClassID = Recipes.RecipeClassID
WHERE 
Recipes.RecipeID IS NULL 
;

-----------------------------------------------------------------
-- "USING" can be used for joining tables on equal values (if the matching columns 
	--in the two tables have the same name- in this case RecipeClassID)

-- "Display the recipe classes that do not yet have any recipes. (USING) "

SELECT Recipe_Classes.RecipeClassDescription
FROM Recipe_Classes 
LEFT OUTER JOIN Recipes 
USING (RecipeClassID)
WHERE Recipes.RecipeID IS NULL
;

SELECT Recipe_Classes.RecipeClassDescription,
	Recipes.RecipeTitle, Recipes.Preparation,
	Ingredients.IngredientName,
	Recipe_Ingredients.RecipeSeqNo,
	Recipe_Ingredients.Amount,
	Measurements.MeasurementDescription
FROM (((Recipe_Classes 
LEFT OUTER JOIN Recipes 
	ON Recipe_Classes.RecipeClassID = 
		Recipes.RecipeClassID)
INNER JOIN Recipe_Ingredients
	ON Recipes.RecipeID = 
	   Recipe_Ingredients.RecipeID)
INNER JOIN Ingredients 
 	ON Ingredients.IngredientID = 
	 Recipe_Ingredients.IngredientID) 
INNER JOIN Measurements 
	ON Measurements.MeasureAmountID = 
	  Recipe_Ingredients.MeasureAmountID 
ORDER BY RecipeTitle, RecipeSeqNo
;
